## 题解


### 思路

用最短路径算法求出每个点x到n号点的最短路径`dis[x]`。
对于受限路径中当前节点为x，下一个节点为y，需要满足`dis[x[>dis[y]`。我们可以按照每个节点的dis值来排序，dis越大的排前面。然后令`f[x]`为从0号点到达x号点的受限路径条数。`f[0] = 1`。遍历排序好的节点x，寻找x所连接的满足`dis[x]>dis[y]`的y，并传递给y贡献x，即`f[y] += f[x]`


### 代码


```cpp
class Solution {
public:
    vector<pair<int,int>> g[20005];
    long dis[20005];
    int vis[20005];
    const long INF = 0x3f3f3f3f3f3f3f3f, M = 1e9+7;
    int countRestrictedPaths(int n, vector<vector<int>>& edges) {
        for (auto&i:edges) {
            g[i[0]].emplace_back(i[1], i[2]);
            g[i[1]].emplace_back(i[0], i[2]);
        }
        memset(dis, 0x3f, sizeof(dis));
        dis[n] = 0;
        priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> que;
        que.emplace(0, n);
        while (que.size()) {
            auto [d, u] = que.top(); que.pop();
            if (vis[u]) continue;
            vis[u] = 1;
            for (auto& [i,j]:g[u]) {
                if (dis[i] > dis[u]+j) {
                    dis[i] = dis[u]+j;
                    que.emplace(dis[i], i);
                }
            }
        }
        // for (int i=1; i<=n; i++) {
        //     cout << dis[i] << " ";
        // }
        vector<long> dp(n+1, 0);
        dp[1] = 1;
        vector<int> id(n, 0);
        for (int i=0; i<n; i++) id[i] = i+1;
        sort(id.begin(), id.end(), [&](auto x, auto y) {return dis[x] > dis[y];});
        int ok = 0;
        for (int i:id) {
            // cout << i << " ";
            if (i == 1) ok = 1;
            if (!ok) continue;
            for (auto [v, d]:g[i]) {
                if (dis[v]<dis[i]) dp[v] += dp[i], dp[v]%=M;
            }
        }
        return dp[n];
    }
};
```
