## 题目

[1106. 逃离大迷宫](https://leetcode.cn/problems/escape-a-large-maze/)


---


在一个 10^6^ x 10^6^ 的网格中，每个网格上方格的坐标为 `(x, y)` 。

现在从源方格 `source = [s_x_, s_y_]` 开始出发，意图赶往目标方格 `target = [t_x_, t_y_]` 。数组 `blocked` 是封锁的方格列表，其中每个 `blocked[i] = [x_i_, y_i_]` 表示坐标为 `(x_i_, y_i_)` 的方格是禁止通行的。

每次移动，都可以走到网格中在四个方向上相邻的方格，只要该方格 **不** 在给出的封锁列表 `blocked` 上。同时，不允许走出网格。

只有在可以通过一系列的移动从源方格 `source` 到达目标方格 `target` 时才返回 `true`。否则，返回 `false`。

 

**示例 1：**

**输入：**blocked = [[0,1],[1,0]], source = [0,0], target = [0,2]
**输出：**false
**解释：**
从源方格无法到达目标方格，因为我们无法在网格中移动。
无法向北或者向东移动是因为方格禁止通行。
无法向南或者向西移动是因为不能走出网格。

**示例 2：**

**输入：**blocked = [], source = [0,0], target = [999999,999999]
**输出：**true
**解释：**
因为没有方格被封锁，所以一定可以到达目标方格。

 

**提示：**

	- `0 <= blocked.length <= 200`

	- `blocked[i].length == 2`

	- `0 <= x_i_, y_i_ < 10^6^`

	- `source.length == target.length == 2`

	- `0 <= s_x_, s_y_, t_x_, t_y_ < 10^6^`

	- `source != target`

	- 题目数据保证 `source` 和 `target` 不在封锁列表内


## 题解


### 思路

离散化，数据范围优化**

封锁列表大小为200 所以 封锁区域是有限的 不超过40000 源方格与目标方格分别广搜在40000步内若队列为空说明被封锁

离散化，每个点的取值的范围很大，但是封锁的点最多200个所以可以离散化出现的点，具体做法就是将出现的点排序映射到一个新的坐标。注意0和1e6以及每个封锁点的周边都需要映射。


### 代码


```cpp
#define ll long long
class Solution {
public:
    int sz = 1e6;
    unordered_set<ll> st;
    struct Node {
        int x, y;
    };
    int dx[4] = {0, 1, 0, -1};
    int dy[4] = {1, 0, -1, 0};

    ll encode(int x, int y) {
        return 1000005LL*x+y;
    }

    int bfs(int x1, int y1, int x2, int y2) {
        queue<Node> que;
        que.push({x1, y1});
        int stp = 0;
        while (!que.empty() && stp < 40000) {
            Node u = que.front(); que.pop();
            if (u.x == x2 && u.y == y2) return 1;
            for (int i=0; i<4; i++) {
                int mx = dx[i]+u.x;
                int my = dy[i]+u.y;
                if (mx<0 || mx>=sz || my<0 || my>=sz || st.count(encode(mx, my)) == 1) continue;
                st.insert(encode(mx, my));
                que.push({mx, my});
            }
            stp++;
        }
        if (que.empty()) return -1;
        return 0;
    }
    bool isEscapePossible(vector<vector<int>>& blocked, vector<int>& source, vector<int>& target) {
        for (int i=0; i<blocked.size(); i++) {
            st.insert(encode(blocked[i][0], blocked[i][1]));
        }
        int s1 = bfs(source[0], source[1], target[0], target[1]);
        if (s1 == 1) return true;
        if (s1 == -1) return false;
        int s2 = bfs(target[0], target[1], source[0], source[1]);
        if (s2 == 1) return true;
        if (s2 == -1) return false;
        return true;
    }
};
//离散化
class Solution {
public:
    bool isEscapePossible(vector<vector<int>>& blocked, vector<int>& source, vector<int>& target) {
        int MX = 1000000;
        map<int,int> r, c;
        r[source[0]]; c[source[1]]; r[target[0]]; c[target[1]];
        r[0]; c[0]; r[MX-1]; c[MX-1];
        for (auto& i:blocked) {
            r[i[0]]; c[i[1]];
            for (int j=0; j<4; j++) {
                int dx = (j-1)%2+i[0];
                int dy = (j-2)%2+i[1];
                if (dx<0 || dx>=MX || dy<0 || dy>=MX) continue;
                r[dx]; c[dy];
            }
        }
        int rsz = 0, csz = 0;
        for (auto& [i, j]:r) {
            j = rsz++;
        }
        for (auto& [i, j]:c) {
            j = csz++;
        }
        queue<pair<int,int>> que;
        que.emplace(r[source[0]], c[source[1]]);
        // cout << rsz << " " << csz << endl;
        vector<vector<int>> g(rsz, vector<int>(csz, 0));
        g[r[source[0]]][c[source[1]]] = 1;
        for (auto& i:blocked) {
            g[r[i[0]]][c[i[1]]] = 1;
        }
        int edx = r[target[0]];
        int edy = c[target[1]];
        // cout << edx << " " << edy << endl;
        while (que.size()) {
            auto [x, y] = que.front(); que.pop();
            if (x == edx && y == edy) return true;
            for (int i=0; i<4; i++) {
                int dx = (i-1)%2+x;
                int dy = (i-2)%2+y;
                if (dx<0 || dx>=rsz || dy<0 || dy>=csz || g[dx][dy]) continue;
                g[dx][dy] = 1;
                que.emplace(dx, dy);
            }
        }
        return false;
    }
};
```
