## 题目

[5. 最长回文子串](https://leetcode.cn/problems/longest-palindromic-substring/)


---


给你一个字符串 `s`，找到 `s` 中最长的 回文 子串。

 

**示例 1：**

**输入：**s = "babad"
**输出：**"bab"
**解释：**"aba" 同样是符合题意的答案。

**示例 2：**

**输入：**s = "cbbd"
**输出：**"bb"

 

**提示：**

	- `1 <= s.length <= 1000`

	- `s` 仅由数字和英文字母组成


## 题解


### 思路

- 区间dp
    
    令`dp[l][r]` 为`s[l...r]`是否是回文串，`dp[l][r]=1`为是， `dp[l][r]=-1`为否。
    初始化`dp[i][i] = 1, dp[i][i+1] = s[l]==[r]?1:-1`
    可见`dp[l][r] = s[l] == s[r] && dp[l+1][r-1] ? 1 : -1`
    
    
    
    
    
- manacher
    
    manacher 算法求最长回文子串。时间复杂度`O(n)`
    预处理+运用dp思想。
    预处理：
    回文串有技术长度有偶数长度，如果用相同的字符插入每个字符之间及首尾，那么所有的回文串长度都会是奇数。
    例如: `abcbcba`
    用`|`分割得 `|a|b|c|b|c|b|a|`
    以回文串的中心字符作为分割点，左右两个子串的长度是相同的，称他们的长度为回文半径。
    dp：
    设$dp_i$为i位置的回文半径。
    r为遍历到当前回文串最远覆盖的下标。可用$r = max(dp_j+j), j \in [0, i-1]$, 这里i是当前所遍历到的下标。m是对应r的中心。
    对于$dp_i$，在i<=r时，分两种情况：
    
    1. 当$dp_i \le r$, 这时$dp_i = dp_j$, j是i关于m对称的点。即$dp_i = dp_{i-2m}$
    2. 当$dp_i > r$, 对于大于r的部分用中心扩散法，所以可以先让$dp_i = r-i$
    由这两种情况可知让$dp_i = min(r-i, dp_{i-2m}$, 然后再做中心扩散法。
    在i>r时，直接使用中心扩散法。
    
    最后便可求得dp数组。以下标i为中心的回文串为[i-dp[i]...dp[i]+i], 这个回文串中去除预处理加的分隔符便是原串的回文子串。

    class Solution {
    public:
        int dp[1005][1005];// dp[l][r] = 1 是回文，-1 不是回文
        int ans = 1;
        string s;
        int dfs(int l, int r) {
            if (dp[l][r]) return dp[l][r];
            dp[l][r] = -1;
            if (s[l] == s[r] && dfs(l+1, r-1) == 1) {
                dp[l][r] = 1;
                ans = max(ans, r-l+1);
            } else {
                if (dfs(l, r-1) == 1) ans = max(ans, r-l);
                if (dfs(l+1, r) == 1) ans = max(ans, r-l);
            }
            return dp[l][r];
        };

            this->s = s;
            int n = s.size();
            
            for (int i=0; i<n; i++) {

            for (int i=0; i+ans<=n; i++) {
                if (dp[i][i+ans-1] == 1) return s.substr(i, ans);

        }

    class Solution { 
    public:
        string longestPalindrome(string s) {
            int n = s.size();
            vector<vector<int>> dp(n, vector<int>(n, 0));// dp[l][r] = 1 是回文，-1 不是回文
            for (int i=0; i<n; i++) {
                dp[i][i] = 1;
                if (i>0 && s[i-1] == s[i]) dp[i-1][i] = 1;
            }
            int ans = 1;
            function<int(int,int)> dfs = [&](int l, int r) {
                if (dp[l][r]) return dp[l][r];
                dp[l][r] = -1;
                if (dfs(l, r-1) == 1) ans = max(ans, r-l);
                if (dfs(l+1, r) == 1) ans = max(ans, r-l);
                if (s[l] == s[r] && dfs(l+1, r-1) == 1) ans = max(ans, r-l+1), dp[l][r] = 1;
                return dp[l][r];
            };
            if (dfs(0, n-1) == 1) return s;
            for (int i=0; i+ans<=n; i++) {
                if (dp[i][i+ans-1] == 1) return s.substr(i, ans);
            }
            return "";
        }
    };

    // 迭代
    // 76 ms
    class Solution {
    public:
        string longestPalindrome(string s) {
            int sz = s.size();
            vector <vector <bool> > dp(sz, vector <bool> (sz,false));
            for (int i=0; i<sz; i++) dp[i][i] = true;
            int maxl = 1;
            int l=0,r=0;
            for (int i=1; i<sz; i++)
                if(s[i] == s[i-1]) {
                    if (2 > maxl) {
                        maxl = 2;
                        l = i-1;
                        r = i;
                    }
                    dp[i-1][i] = true;
                }
            for (int i=0, j=0; (i+j) <= 2*(sz-1); (i+j)%2 ? i++ : j++) {
                int a = i, b = j;
                while ( a-1 >= 0 && b+1 < sz && s[a-1] == s[b+1] && dp[a][b]) {
                    dp[--a][++b] = true;
                    if (b-a+1 > maxl) {
                        maxl = b-a+1;
                        l = a;
                        r = b;
                    }
                }
            }
            return s.substr(l,r-l+1);
        }
    };

    class Solution {
    public:
        int dp[2005];
        string longestPalindrome(string s) {
            string str = "*";
            for (char i:s) {
                str.push_back(i);
                str.push_back('*');
            }
            int r = 0, m = 0;
            for (int i=1; i<str.size(); i++) {
                if (i<=r) dp[i] = min(dp[i-2*m], r-i);
                int l = i-dp[i]-1, r = i+dp[i]+1;
                while (l>=0 && r<str.size() && str[l] == str[r]) {
                    l--, r++;
                    dp[i]++;
                }
                if (dp[i]+i > r) {
                    r = dp[i]+i;
                    m = i;
                }
            }
            int mx = *max_element(dp, dp+str.size());
            for (int i=0; i<str.size(); i++) {
                if (mx == dp[i]) {
                    string rt;
                    for (int j=i-dp[i]; j<=i+dp[i]; j++) {
                        if (str[j] != '*') {
                            rt.push_back(str[j]);
                        }
                    }
                    return rt;
                }
                // cout << i << " " << str[i] << " " << dp[i] << endl;
            }
            return str;
        }
    };
