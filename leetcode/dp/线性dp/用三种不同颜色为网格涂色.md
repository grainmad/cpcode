## 题目

[2061. 用三种不同颜色为网格涂色](https://leetcode.cn/problems/painting-a-grid-with-three-different-colors/)


---


给你两个整数 `m` 和 `n` 。构造一个 `m x n` 的网格，其中每个单元格最开始是白色。请你用 **红、绿、蓝** 三种颜色为每个单元格涂色。所有单元格都需要被涂色。

涂色方案需要满足：**不存在相邻两个单元格颜色相同的情况** 。返回网格涂色的方法数。因为答案可能非常大， 返回 **对 **`10^9^ + 7`** 取余** 的结果。

## 题解


### 思路

令`dp[c[i]][j]`为以m位相邻位不同的三进制数`c[i]`作为结尾在前j列的组合数。
对于`m * 1`的网格的组合数是`3 * 2^(m-1)`, 而m最多是5，所以每一列的合法种数最多不过48
则有`dp[c[j]][i] = sum(dp[c[k]][i-1]), k from 0 to m-1`,且`c[j]`与`c[k]`每个三进制对应位不同。
实际上`dp[][i-1]` 到 `dp[][i]` 的转移可以用一个矩阵表示。可以使用矩阵快速幂优化时间复杂度，但是n的取值最多1000所以效果不明显。


### 代码


```cpp
class Solution {
public:
    vector<int> c;
    long dp[50][1005];
    void ger(int cur, int lev) {
        if (lev == 0) {
            c.push_back(cur);
            return ;
        }
        for (int i=0; i<3; i++) {
            if (cur%3 == i) continue;
            ger(cur*3+i, lev-1);
        }
    }
    int colorTheGrid(int m, int n) {
        ger(0, m-1);
        ger(1, m-1);
        ger(2, m-1);
        int MOD = 1e9+7;
        // cout << c.size() << endl;
        auto ok = [&](int a, int b)->bool {
            for (int i=0; i<m; i++) {
                if (a%3 == b%3) return false;
                a /= 3;
                b /= 3;
            }
            return true;
        };
        for (int i=0; i<c.size(); i++) dp[i][0] = 1;
        for (int i=1; i<n; i++) {
            for (int j=0; j<c.size(); j++) {
                for (int k=0; k<c.size(); k++) {
                    if (ok(c[j], c[k])) dp[j][i] += dp[k][i-1], dp[j][i] %= MOD;
                }
            }
        }
        long ans = 0;
        for (int i=0; i<c.size(); i++) {
            ans += dp[i][n-1];
            ans %= MOD;
        }
        return ans;
    }
};

//matrix
class Solution {
public:
    #define ll long long
    const int MOD = 1e9+7;
    vector<int> c;
    struct Matrix {
        ll mat[50][50];
    };
    // a: n*n b: n*n
    Matrix mul_matrix(Matrix a, Matrix b, int n) { 
        Matrix res;
        for (int i=0; i<n; i++) {
            for (int j=0; j<n; j++) {
                res.mat[i][j] = 0;
                for (int k=0; k<n; k++) {
                    res.mat[i][j] += a.mat[i][k]*b.mat[k][j];
                    res.mat[i][j] %= MOD;
                }
            }
        }
        return res;
    }
    //a^p
    Matrix pow_matrix(Matrix a, int p, int n) {
        Matrix res;
        //单位矩阵 
        for (int i=0; i<n; i++) {
            for (int j=0; j<n; j++) {
                res.mat[i][j] = (i == j? 1 : 0);
            }
        }
        while (p != 0) {
            if (p&1) res = mul_matrix(a, res, n);
            a = mul_matrix(a, a, n);
            p >>= 1;
        }
        return res;
    }
    void ger(int cur, int lev) {
        if (lev == 0) {
            c.push_back(cur);
            return ;
        }
        for (int i=0; i<3; i++) {
            if (cur%3 == i) continue;
            ger(cur*3+i, lev-1);
        }
    }
    int colorTheGrid(int m, int n) {
        ger(0, m-1);
        ger(1, m-1);
        ger(2, m-1);
        // cout << c.size() << endl;
        auto ok = [&](int a, int b)->bool {
            for (int i=0; i<m; i++) {
                if (a%3 == b%3) return false;
                a /= 3;
                b /= 3;
            }
            return true;
        };
        Matrix A, B;
        for (int j=0; j<c.size(); j++) {
            for (int k=0; k<c.size(); k++) {
                A.mat[j][k] = ok(c[j], c[k])?1:0;
                B.mat[j][k] = 0;
            }
            B.mat[j][0] = 1;
        }
        A = pow_matrix(A, n-1, c.size());
        B = mul_matrix(A, B, c.size());
        long ans = 0;
        for (int i=0; i<c.size(); i++) {
            ans += B.mat[i][0];
            ans %= MOD;
        }
        return ans;
    }
};
```
